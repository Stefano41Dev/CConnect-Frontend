═══════════════════════════════════════════════════════════════════════════════

ABRIR/CERRAR:
   this.postModal?.openModal()    // Abre el modal
   this.postModal?.closeModal()   // Cierra

IMÁGENES:
   this.postModal?.removeImage(0)     // Elimina imagen
   this.postModal?.triggerFileInput() // Abre selector

VALIDACIÓN:
   this.postModal?.isFormValid()      // Retorna boolean

OTRAS:
   this.postModal?.resetForm()        // Limpia todo
   this.postModal?.submitPost()       // Envía (AGREGAR API)
